# 分层架构落地

> 学完本节，你能回答：Controller-Service-Repository 三层各自守什么边界？为什么 Service 不该直接操作 HTTP，Repository 不该包含业务规则？在 Python 中如何用最小依赖落地这一分层？

## 为什么要分层

不分层的“面条代码”有三个痛点：

1. **测试贵**：改一行业务要端到端起服务、连数据库、调外部接口。
2. **替换难**：换存储（如从 SQLite 到 Postgres）要改遍业务代码。
3. **定位慢**：一次请求出错，分不清是校验、业务还是存储的问题。

分层用**单向依赖**解决：`Controller → Service → Repository`，上层依赖下层，下层不感知上层。每层只对自己的抽象负责，替换与测试都沿边界进行。

- **Controller（表现层）**：HTTP 的翻译官。负责路由、参数解析（Pydantic）、状态码、错误体，不含业务规则。薄而可测。
- **Service（业务层）**：领域的编排者。持有业务规则与事务边界，通过 Repository 接口操作数据，不感知 HTTP。
- **Repository（数据层）**：存储的抽象。封装 SQL / ORM / 文件，提供 `create / get / update` 等领域友好的方法，隐藏持久化细节。

> **跨平台提示**：分层本身与平台无关；若示例涉及文件路径，统一用 `pathlib.Path` 兼容 Windows（`\`）与 macOS / Linux（`/`）。

## 依赖方向与测试策略

```
HTTP Request
    ↓
Controller (FastAPI router) ——— 薄，测路由与校验
    ↓ 依赖注入 Service
Service ——— 厚，测业务规则（可 mock Repository）
    ↓ 依赖注入 Repository
Repository (TaskStore) ——— 测持久化（可用临时库）
    ↓
SQLite / Postgres / 内存
```

- Controller 测试用 `TestClient` 覆盖路由与校验，不连真实数据库（可注入 Fake Repository）。
- Service 测试可 mock Repository，聚焦业务分支。
- Repository 测试用真实或临时存储，验证 SQL 与事务。

这种分层在 MeetingToText 的 `backend/app` 中有更完整的落地，本节用单文件最小实现让边界可见。

## 可运行示例：router → service → repo 分层模块

文件 `book/part2_backend_development/chapter03_backend_essence/demo_layered.py`（单文件演示三层，`TestClient` 本地验证，无网络）：

```{code-cell} ipython3
# 文件 book/part2_backend_development/chapter03_backend_essence/demo_layered.py
import tempfile, pathlib
from typing import Protocol

from fastapi import FastAPI, HTTPException
from fastapi.testclient import TestClient
from pydantic import BaseModel, Field
from m2t.store import TaskStore

# --- Repository 层：定义接口与两种实现（真实与内存） ---
class TaskRepository(Protocol):
    def create(self, task_id: str, filename: str) -> None: ...
    def get(self, task_id: str) -> dict | None: ...
    def list_all(self) -> list[dict]: ...

class SqliteTaskRepository:
    """基于 m2t.store.TaskStore 的真实实现（教学复用）"""
    def __init__(self, db_path: pathlib.Path):
        self.store = TaskStore(db_path)
    def create(self, task_id: str, filename: str) -> None:
        self.store.create(task_id, filename)
    def get(self, task_id: str) -> dict | None:
        return self.store.get(task_id)
    def list_all(self) -> list[dict]:
        return self.store.list_tasks(limit=100)

class InMemoryTaskRepository:
    """内存实现（用于演示替换，测试中常用）"""
    def __init__(self):
        self._data: dict[str, dict] = {}
    def create(self, task_id: str, filename: str) -> None:
        self._data[task_id] = {"id": task_id, "filename": filename, "status": "pending"}
    def get(self, task_id: str) -> dict | None:
        return self._data.get(task_id)
    def list_all(self) -> list[dict]:
        return list(self._data.values())

# --- Service 层：业务规则，不感知 HTTP ---
class TaskService:
    def __init__(self, repo: TaskRepository):
        self.repo = repo

    def create_task(self, task_id: str, filename: str) -> dict:
        if not task_id or not filename:
            raise ValueError("task_id and filename must be non-empty")
        if self.repo.get(task_id) is not None:
            raise ValueError("already exists")
        self.repo.create(task_id, filename)
        row = self.repo.get(task_id)
        assert row is not None
        return row

    def get_task(self, task_id: str) -> dict:
        row = self.repo.get(task_id)
        if row is None:
            raise KeyError("not found")
        return row

# --- Controller 层：HTTP 翻译，薄 ---
class CreateIn(BaseModel):
    task_id: str = Field(min_length=1)
    filename: str = Field(min_length=1)

class CreateOut(BaseModel):
    id: str
    filename: str
    status: str

def build_app(repo: TaskRepository) -> FastAPI:
    service = TaskService(repo)
    app = FastAPI(title="Layered Demo")

    @app.post("/tasks", response_model=CreateOut)
    def create(payload: CreateIn):
        try:
            row = service.create_task(payload.task_id, payload.filename)
        except ValueError as e:
            if "already exists" in str(e):
                raise HTTPException(status_code=409, detail=str(e))
            raise HTTPException(status_code=422, detail=str(e))
        return CreateOut(id=row["id"], filename=row["filename"], status=row["status"])

    @app.get("/tasks/{task_id}", response_model=CreateOut)
    def get_one(task_id: str):
        try:
            row = service.get_task(task_id)
        except KeyError:
            raise HTTPException(status_code=404, detail="not found")
        return CreateOut(id=row["id"], filename=row["filename"], status=row["status"])

    @app.get("/tasks")
    def list_tasks():
        return repo.list_all()

    return app

# --- 验证 1：用内存实现（不碰文件，适合单测） ---
mem_repo = InMemoryTaskRepository()
app_mem = build_app(mem_repo)
client_mem = TestClient(app_mem)

r1 = client_mem.post("/tasks", json={"task_id": "m1", "filename": "a.wav"})
print("mem create m1:", r1.json())
assert r1.status_code == 200

r2 = client_mem.post("/tasks", json={"task_id": "m1", "filename": "a.wav"})
print("mem duplicate:", r2.status_code)
assert r2.status_code == 409

r3 = client_mem.get("/tasks/m1")
print("mem get m1:", r3.json())
assert r3.json()["id"] == "m1"

# --- 验证 2：用 SQLite 实现（复用 m2t.store，适合集成测试） ---
with tempfile.TemporaryDirectory() as td:
    db_path = pathlib.Path(td) / "layered.db"
    sqlite_repo = SqliteTaskRepository(db_path)
    app_sql = build_app(sqlite_repo)
    client_sql = TestClient(app_sql)

    s1 = client_sql.post("/tasks", json={"task_id": "s1", "filename": "meeting.wav"})
    print("sqlite create s1:", s1.json())
    assert s1.status_code == 200

    s2 = client_sql.get("/tasks/s1")
    print("sqlite get s1:", s2.json())
    assert s2.json()["filename"] == "meeting.wav"

    s3 = client_sql.get("/tasks")
    print("sqlite list:", s3.json())
    assert len(s3.json()) == 1

    # Service 层可脱离 HTTP 直接测试（演示分层价值）
    svc = TaskService(sqlite_repo)
    row = svc.get_task("s1")
    print("service direct get:", row["id"], row["filename"])
    assert row["id"] == "s1"

print("分层校验通过：Controller 薄、Service 厚、Repository 可替换")
# 预期输出:
# mem create m1: {'id': 'm1', 'filename': 'a.wav', 'status': 'pending'}
# mem duplicate: 409
# mem get m1: {'id': 'm1', ...}
# sqlite create s1: {'id': 's1', ...}
# sqlite get s1: {'id': 's1', ...}
# sqlite list: [{'id': 's1', ...}]
# service direct get: s1 meeting.wav
# 分层校验通过：Controller 薄、Service 厚、Repository 可替换
```
