# 关系型数据库原理

> 学完本节，你能回答：为什么“存进去的数据”不等同于“存对了的数据”？事务如何保证并发与故障下的正确性？索引为什么能加速查询却拖慢写入？锁与 WAL 又在什么层面解决并发冲突？

## 先动手：WAL 与 busy_timeout 如何让并发写入不直接报错

别先背事务术语，先看两个工程参数的实测。下面的 Demo 用 `sqlite3` 与多线程并发写入对比 `DELETE` 模式与 `WAL` 模式，并观测 `busy_timeout` 是否让锁冲突自动等待而非抛错。结果会直接告诉你持久性与隔离性在解决什么。

```{code-cell} ipython3
import sqlite3
import tempfile
import pathlib
import threading
import time

tmpdir = tempfile.TemporaryDirectory()
db_path = pathlib.Path(tmpdir.name) / "wal_demo.db"

def init_db(journal_mode: str = "WAL", busy_timeout_ms: int = 5000):
    conn = sqlite3.connect(str(db_path), timeout=busy_timeout_ms / 1000, isolation_level=None, check_same_thread=False)
    conn.execute(f"PRAGMA journal_mode={journal_mode}")
    conn.execute(f"PRAGMA busy_timeout={busy_timeout_ms}")
    conn.execute("""
    CREATE TABLE IF NOT EXISTS tasks (
        id TEXT PRIMARY KEY,
        filename TEXT NOT NULL,
        status TEXT NOT NULL DEFAULT 'pending'
    )
    """)
    conn.commit()
    return conn

# 对照组：WAL 加等待，让并发写入可被串行化
conn = init_db("WAL", 5000)
print("journal_mode:", conn.execute("PRAGMA journal_mode").fetchone()[0])
print("busy_timeout:", conn.execute("PRAGMA busy_timeout").fetchone()[0])

def concurrent_writes(n: int = 8):
    errors: list[str] = []
    def _insert(i: int):
        try:
            c = sqlite3.connect(str(db_path), timeout=5, isolation_level=None, check_same_thread=False)
            c.execute("PRAGMA journal_mode=WAL")
            c.execute("PRAGMA busy_timeout=5000")
            c.execute("BEGIN IMMEDIATE")
            c.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", (f"t{i}", f"f{i}.wav"))
            # 模拟持有写锁一小段时间，让其他线程触发 busy
            time.sleep(0.02)
            c.execute("COMMIT")
            c.close()
        except Exception as e:
            errors.append(str(e))
    threads = [threading.Thread(target=_insert, args=(i,)) for i in range(n)]
    for th in threads:
        th.start()
    for th in threads:
        th.join()
    return errors

errs = concurrent_writes(8)
cnt = conn.execute("SELECT COUNT(*) FROM tasks").fetchone()[0]
print("after concurrent inserts, count:", cnt, "errors:", len(errs))
assert cnt == 8
assert len(errs) == 0

# 崩溃恢复的体感：WAL 先写日志再写数据文件，COMMIT 后即持久
conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t_final", "final.wav"))
conn.commit()
# 重开连接验证持久性
conn2 = sqlite3.connect(str(db_path))
cnt2 = conn2.execute("SELECT COUNT(*) FROM tasks").fetchone()[0]
print("reopened count:", cnt2)
assert cnt2 == 9
conn.close()
conn2.close()
tmpdir.cleanup()
print("WAL 校验通过：并发写入可等待串行化，提交后可持久重读")
# 预期输出:
# journal_mode: wal
# busy_timeout: 5000
# after concurrent inserts, count: 8 errors: 0
# reopened count: 9
# WAL 校验通过：...
```

观测小结：`WAL` 让读不阻塞写，`busy_timeout=5000` 让短暂写冲突等待而非直接抛 `database is locked`。`m2t/store.py` 的 `TaskStore._get_conn()` 中两行 `PRAGMA` 正是这一结论的工程落地。

再看事务的原子性如何自证：

```{code-cell} ipython3
import sqlite3

conn = sqlite3.connect(":memory:")
conn.execute("PRAGMA journal_mode=WAL")
conn.execute("""
CREATE TABLE tasks (
    id TEXT PRIMARY KEY,
    filename TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'pending',
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    full_text TEXT DEFAULT ''
)
""")
conn.execute("CREATE INDEX idx_tasks_created_at ON tasks(created_at DESC)")
conn.commit()

# 提交：两条插入在同一事务中一并可见
conn.execute("BEGIN")
conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t1", "a.wav"))
conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t2", "b.wav"))
conn.commit()
cnt = conn.execute("SELECT COUNT(*) FROM tasks").fetchone()[0]
print("after commit:", cnt)
assert cnt == 2

# 回滚：主键冲突触发异常，显式 ROLLBACK 回到起点
try:
    conn.execute("BEGIN")
    conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t3", "c.wav"))
    conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t1", "dup.wav"))
    conn.commit()
except sqlite3.IntegrityError as e:
    print("integrity error:", e)
    conn.rollback()

cnt2 = conn.execute("SELECT COUNT(*) FROM tasks").fetchone()[0]
print("after rollback:", cnt2)
assert cnt2 == 2

# 保存点：事务内的子事务，可局部回滚
conn.execute("BEGIN")
conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t3", "c.wav"))
conn.execute("SAVEPOINT sp1")
conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t4", "d.wav"))
conn.execute("ROLLBACK TO SAVEPOINT sp1")
conn.execute("RELEASE SAVEPOINT sp1")
conn.commit()
rows = conn.execute("SELECT id FROM tasks ORDER BY id").fetchall()
print("ids after savepoint:", [r[0] for r in rows])
assert [r[0] for r in rows] == ["t1", "t2", "t3"]

conn.close()
print("事务校验通过：提交与回滚与保存点均按预期")
# 预期输出:
# after commit: 2
# integrity error: UNIQUE constraint failed: tasks.id
# after rollback: 2
# ids after savepoint: ['t1', 't2', 't3']
# 事务校验通过：...
```

## 从实验命名原理：事务、索引与锁

刚才的两次实测已把原理跑出来了，再补命名：

事务是把多条 SQL 打包为“要么全做，要么全不做”的原子单元，`BEGIN` 与 `COMMIT` 划定边界，失败则 `ROLLBACK`。这四条约束常被概括为原子性、一致性、隔离性、持久性。原子性保证打包内全生效或全不生效，一致性保证约束如主键与非空始终成立，隔离性描述并发事务互不干扰的程度，持久性保证提交后落盘并可通过 WAL 恢复。SQLite 通过锁与 WAL 提供近似的串行化，PostgreSQL 与 MySQL 则提供多档隔离级别可选。

索引是“用空间换时间的目录”。`m2t/store.py` 的 `idx_tasks_created_at` 就是为 `ORDER BY created_at DESC` 加的目录。B-Tree 索引适合等值与范围查询，代价是写入时需同步更新，所以高频 `WHERE` 或 `ORDER BY` 或 `JOIN` 列适合建索引，低区分度列则不必。

锁同样可从实验反推。读可并发，写需独占。SQLite 是单写者多读者，WAL 下读不阻塞写。乐观锁假设冲突少、提交时检查，悲观锁假设冲突多、操作前先加锁，选型看冲突概率与重试成本。
