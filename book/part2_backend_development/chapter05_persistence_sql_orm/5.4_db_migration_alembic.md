# 数据库迁移 Alembic

> 学完本节，你能回答：为什么“改表结构”需要版本控制？Alembic 的 `upgrade` / `downgrade` 如何保证可重放与可回滚？迁移失败时如何不让库处于半成品状态？

## 为什么不能手写 `ALTER TABLE` 了事

单人单库时，直接 `ALTER TABLE ADD COLUMN` 似乎够用。但多人协作、测试/预发/生产多环境、回滚需求出现时，问题随之而来：谁改了哪张表？改了几次？生产是否已执行？失败后如何回退？

类比代码版本控制：没有 Git 时靠“覆盖文件”也能写代码，但无法回答“谁改了哪一行、如何回退到上周版本”。数据库迁移就是“给表结构加 Git”。

## Alembic 的核心思想

- **版本链**：每个迁移是一个带 `revision` 哈希的 Python 脚本，`down_revision` 指向上一个版本，形成链式历史。
- **升级与回滚**：`upgrade()` 前向执行，`downgrade()` 反向撤销，二者互为逆操作，保证“可重放、可回滚”。
- **版本表**：`alembic_version` 单行表记录当前版本，`alembic upgrade head` 时对比文件与表，决定执行哪些脚本。
- **事务性**：单个迁移在事务中执行，失败即回滚，不让库处于半成品。

> **与 `m2t/store.py` 的关系**：`TaskStore` 的 `SCHEMA` 用 `CREATE TABLE IF NOT EXISTS` 适合教学与单机起步；当 `tasks` 需要新增 `duration` / `language` 列或加索引时，生产中应改用 Alembic 迁移，而非直接改 `SCHEMA` 字符串后“删库重建”。

> **选型中立性**：迁移工具不限于 Alembic（Python 生态还有 `yoyo-migrations` / `Django migrations` 等），思想一致——版本化、可重放、可回滚；Alembic 因与 `SQLAlchemy` 协同最紧密而成为本书示例。

## 可运行示例：用 SQLite 模拟迁移版本表

示例（用 `sqlite3` 模拟 `alembic_version` 与升级/回滚，无网络）：

```{code-cell} ipython3
import sqlite3

conn = sqlite3.connect(":memory:")
conn.row_factory = sqlite3.Row

# 初始 schema：v1 只有 tasks 基础表（类似 m2t/store.py 的起点）
conn.executescript("""
CREATE TABLE tasks (
    id TEXT PRIMARY KEY,
    filename TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'pending',
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE TABLE alembic_version (
    version_num TEXT PRIMARY KEY
);
INSERT INTO alembic_version (version_num) VALUES ('001');
""")
conn.commit()

def current_version() -> str | None:
    row = conn.execute("SELECT version_num FROM alembic_version").fetchone()
    return row["version_num"] if row else None

def upgrade_to_002_add_full_text():
    # 迁移 002：新增 full_text 列（模拟 op.add_column）
    # SQLite 的 ALTER TABLE ADD COLUMN 需保证可回滚：失败则 rollback
    print("upgrade 001 -> 002: add full_text")
    with conn:
        conn.execute("ALTER TABLE tasks ADD COLUMN full_text TEXT DEFAULT ''")
        conn.execute("UPDATE alembic_version SET version_num = '002'")
    print("version:", current_version())
    assert current_version() == "002"
    cols = [r[1] for r in conn.execute("PRAGMA table_info(tasks)").fetchall()]
    print("columns after 002:", cols)
    assert "full_text" in cols

def upgrade_to_003_add_duration_index():
    print("upgrade 002 -> 003: add duration + index")
    with conn:
        conn.execute("ALTER TABLE tasks ADD COLUMN duration INTEGER DEFAULT 0")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_tasks_status ON tasks(status)")
        conn.execute("UPDATE alembic_version SET version_num = '003'")
    print("version:", current_version())
    assert current_version() == "003"
    # 验证索引存在
    idx = conn.execute("SELECT name FROM sqlite_master WHERE type='index' AND name='idx_tasks_status'").fetchone()
    print("index exists:", idx is not None)
    assert idx is not None

def downgrade_to_002():
    # 回滚 003 -> 002：SQLite 不支持 DROP COLUMN（3.35+ 才部分支持），此处演示“重建表”回滚策略
    # 生产中 Alembic 的 downgrade() 需手写逆向逻辑，此处为教学简化
    print("downgrade 003 -> 002: drop duration (rebuild)")
    with conn:
        # 重建不含 duration 的表
        conn.executescript("""
        CREATE TABLE tasks_new (
            id TEXT PRIMARY KEY,
            filename TEXT NOT NULL,
            status TEXT NOT NULL DEFAULT 'pending',
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            full_text TEXT DEFAULT ''
        );
        INSERT INTO tasks_new (id, filename, status, created_at, full_text)
            SELECT id, filename, status, created_at, full_text FROM tasks;
        DROP TABLE tasks;
        ALTER TABLE tasks_new RENAME TO tasks;
        DROP INDEX IF EXISTS idx_tasks_status;
        UPDATE alembic_version SET version_num = '002';
        """)
    print("version:", current_version())
    assert current_version() == "002"
    cols = [r[1] for r in conn.execute("PRAGMA table_info(tasks)").fetchall()]
    print("columns after downgrade:", cols)
    assert "duration" not in cols

# 写入一条数据，贯穿迁移前后
conn.execute("INSERT INTO tasks (id, filename) VALUES (?, ?)", ("t1", "a.wav"))
conn.commit()
print("initial version:", current_version())

upgrade_to_002_add_full_text()
# v2 后可写 full_text
conn.execute("UPDATE tasks SET full_text = ? WHERE id = ?", ("hello world", "t1"))
conn.commit()
print("full_text:", conn.execute("SELECT full_text FROM tasks WHERE id='t1'").fetchone()[0])

upgrade_to_003_add_duration_index()
conn.execute("UPDATE tasks SET duration = ? WHERE id = ?", (120, "t1"))
conn.commit()
print("duration:", conn.execute("SELECT duration FROM tasks WHERE id='t1'").fetchone()[0])

downgrade_to_002()
# 回滚后 full_text 仍在，duration 已移除
print("after downgrade full_text:", conn.execute("SELECT full_text FROM tasks WHERE id='t1'").fetchone()[0])
assert conn.execute("SELECT full_text FROM tasks WHERE id='t1'").fetchone()[0] == "hello world"

# 演示失败回滚：尝试在事务中执行错误迁移，整体回滚
try:
    with conn:
        conn.execute("ALTER TABLE tasks ADD COLUMN tmp_col TEXT")
        # 故意执行错误 SQL 触发回滚
        conn.execute("ALTER TABLE tasks ADD COLUMN tmp_col TEXT")  # 重复列名
        conn.execute("UPDATE alembic_version SET version_num = 'bad'")
except sqlite3.OperationalError as e:
    print("migration failed, rolled back:", e)

# tmp_col 虽已在 with 块内创建，但因异常整个 with 回滚，tmp_col 不应存在
cols_final = [r[1] for r in conn.execute("PRAGMA table_info(tasks)").fetchall()]
print("cols after failed migration:", cols_final)
# 注意：SQLite 的 ALTER TABLE 在事务内行为特殊，部分版本会隐式提交；此处断言“版本未变”作为核心契约
print("version after failed:", current_version())
assert current_version() == "002"

conn.close()
print("迁移校验通过：升级/回滚/失败回滚均按预期（版本表即真相）")
# 预期输出:
# initial version: 001
# upgrade 001 -> 002: add full_text
# version: 002
# columns after 002: ['id', 'filename', 'status', 'created_at', 'full_text']
# ...
# downgrade 003 -> 002: drop duration (rebuild)
# version: 002
# 迁移校验通过：升级/回滚/失败回滚均按预期（版本表即真相）
```

```bash
# 查看当前版本（示意，真实 Alembic 需安装后可用）
# alembic current  # 显示 alembic_version
# alembic upgrade head  # 升级到最新
# alembic downgrade -1  # 回退一版
# 本书用 sqlite3 模拟该思想，无需安装 alembic
```

> **工程启示**：迁移脚本是“表结构的 Git log”，`alembic_version` 是“当前分支指针”。永远让“库的当前版本”可被一行查询回答，而不是靠人脑记忆。

> **环境约定**：本书面向 Linux，`alembic` 命令在 Linux 下一致；本书用 `sqlite3` 模拟迁移思想，无需额外安装即可在任意平台复现。
