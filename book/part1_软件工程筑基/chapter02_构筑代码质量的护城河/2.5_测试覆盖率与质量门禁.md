# 测试覆盖率与质量门禁

> 覆盖率是“后视镜”，不是“方向盘”——它告诉你哪些路没走过，但不告诉你路走得对不对。门禁的价值在于“守住底线”，而非“追逐数字”。

## 覆盖率：度量什么、忽略什么

- **行覆盖（Line Coverage）**：有多少行被执行过。易度量，但无法区分“执行了”与“正确了”。
- **分支覆盖（Branch Coverage）**：有多少 `if/else` 分支被走到。比行覆盖更能暴露“异常路径未测”。
- **误用**：100% 行覆盖仍可能遗漏“错误输入的防御分支”（如 `TaskStore.get` 返回 `None` 的分支未断言）。

对 `m2t` 而言，`m2t/export.py` 的 `_get_segments` 有 `dict` vs 对象的两条分支，若只测 `dict` 形态，行覆盖可能是 100%，分支覆盖却缺一侧。

## CI 红线：把门禁写进流水线

质量门禁应在 CI 中自动化、一致化，而非依赖人工“记得跑”：

```yaml
# .github/workflows/ci.yml 片段（示意，非真实执行）
# Windows / macOS / Linux 均由 runner 决定，路径用 pathlib 兼容
- name: Lint & Format
  run: |
    .venv/bin/ruff check m2t/          # 或 .venv\Scripts\ruff.exe check m2t\
    .venv/bin/ruff format --check m2t/
- name: Type check
  run: .venv/bin/mypy m2t/
- name: Tests with coverage
  run: .venv/bin/pytest --cov=m2t --cov-report=term --cov-fail-under=80
```

- `--cov-fail-under=80` 是红线：低于 80% 则 CI 失败，阻止合并。
- 门禁应为**增量门禁**（如“新代码覆盖率不低于 80%”）而非“一刀切 100%”，避免为数字而测试。
- 本地与 CI 共用同一 `pyproject.toml`，保证规则一致。

> **中立性说明**：覆盖率阈值无银弹。80% 是教学仓的演示值，真实团队应根据“缺陷成本 × 变更频率”设定；对关键模块（如 `store` 的事务边界）可设更高阈值，对实验性代码可适度放宽。

## 可运行示例：覆盖率的度量思路

文件 `book/part1_软件工程筑基/chapter02_构筑代码质量的护城河/demo_coverage_measure.py`（用标准库 `trace` 与手工计数演示“覆盖率如何算”）：

```{code-cell} ipython3
# 文件 book/part1_软件工程筑基/chapter02_构筑代码质量的护城河/demo_coverage_measure.py
import pathlib, textwrap, tempfile, sys, trace, importlib.util

# 构造一个待测模块（模拟 m2t/export 的分支逻辑）
module_code = textwrap.dedent("""
def export(task, fmt):
    fmt = fmt.lower().strip()
    if fmt == "txt":
        if not task.get("segments"):
            return task.get("full_text", "")
        return "\\n".join(s["text"] for s in task["segments"])
    elif fmt == "srt":
        if not task.get("segments"):
            return ""
        return "srt:" + str(len(task["segments"]))
    else:
        raise ValueError(f"bad fmt: {fmt}")
""")

with tempfile.TemporaryDirectory() as td:
    mod_path = pathlib.Path(td) / "sample_export.py"
    mod_path.write_text(module_code, encoding="utf-8")

    # 用例集 A：仅覆盖 txt 正常路径 + srt 空分支
    cases_a = [
        ({"segments": [{"text": "hi"}], "full_text": ""}, "txt"),
        ({"segments": [], "full_text": ""}, "srt"),
    ]
    # 用例集 B：覆盖 txt 空回退 + srt 正常 + 异常路径
    cases_b = [
        ({"segments": [], "full_text": "fallback"}, "txt"),
        ({"segments": [{"text": "a"}, {"text": "b"}]}, "srt"),
        ({"segments": []}, "unknown"),
    ]

    def measure(cases):
        spec = importlib.util.spec_from_file_location("sample_export", str(mod_path))
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
        tracer = trace.Trace(count=True, trace=False)
        for task, fmt in cases:
            try:
                tracer.runfunc(mod.export, task, fmt)
            except ValueError:
                pass
        # 统计覆盖：trace 计数包含每行执行次数
        counts = tracer.counts
        # 仅统计本文件的行
        covered = {(fname, lineno) for (fname, lineno), c in counts.items() if pathlib.Path(fname).name == "sample_export.py"}
        total_lines = len([l for l in module_code.splitlines() if l.strip() and not l.strip().startswith("#")])
        # 简化：covered 行数即 tracer 记录的行数
        print(f"cases={len(cases)} covered_lines={len(covered)} total≈{total_lines}")
        print("covered keys:", sorted(list(covered))[:10], "...")
        return len(covered)

    print("=== 用例集 A（部分覆盖）===")
    cov_a = measure(cases_a)
    print("=== 用例集 B（更充分）===")
    cov_b = measure(cases_b)
    print(f"覆盖对比: A={cov_a} < B={cov_b} ? {cov_a < cov_b}")
    print("启示：行覆盖可量化，但仅靠 A 仍遗漏 txt 空回退与异常分支")
# 预期输出:
# === 用例集 A（部分覆盖）===
# cases=2 covered_lines=... total≈...
# === 用例集 B（更充分）===
# cases=3 covered_lines=... (> A)
# 覆盖对比: A=... < B=... ? True
```

文件 `book/part1_软件工程筑基/chapter02_构筑代码质量的护城河/demo_coverage_m2t.py`（对 `m2t` 真实模块的覆盖率体感，若安装 `coverage` 则走真实度量，否则回退到手工计数）：

```{code-cell} ipython3
# 文件 book/part1_软件工程筑基/chapter02_构筑代码质量的护城河/demo_coverage_m2t.py
import subprocess, sys, pathlib, textwrap, tempfile

# 优先尝试 coverage（pytest-cov 依赖）；若无则用 trace 回退，保证可复现
try:
    import coverage  # type: ignore
    has_cov = True
except ImportError:
    has_cov = False

print("has_coverage:", has_cov)

if has_cov:
    # 用 coverage 测量 m2t.export 的真实覆盖
    cov = coverage.Coverage(source=["m2t"], branch=True)
    cov.start()
    from m2t.export import export
    # 执行一组代表性用例（覆盖正常、空、异常）
    export({"filename": "a.wav", "result": {"segments": [{"speaker": "说话人1", "text": "hi", "start": 0, "end": 1}]}}, "txt")
    export({"filename": "b.wav", "result": {"segments": []}, "full_text": "fb"}, "txt")
    export({"filename": "c.wav", "result": {"segments": [{"speaker": "", "text": "x", "start": -1, "end": 0}]}}, "srt")
    try:
        export({}, "bad")
    except ValueError:
        pass
    cov.stop()
    print("coverage report (m2t/export.py):")
    cov.report(show_missing=False, include="m2t/export.py*")
    print("branch coverage enabled:", cov.config.branch)
else:
    # 回退：手工统计 m2t.export 的关键分支是否被走到
    from m2t.export import export
    branches_hit = set()
    # txt 分支
    out = export({"filename": "a.wav", "result": {"segments": [{"speaker": "说话人1", "text": "hi", "start": 0, "end": 1}]}}, "txt")
    branches_hit.add("txt_normal")
    out2 = export({"filename": "b.wav", "result": {"segments": []}, "full_text": "fb"}, "txt")
    branches_hit.add("txt_empty_fallback")
    out3 = export({"filename": "c.wav", "result": {"segments": [{"speaker": "", "text": "x", "start": -1, "end": 0}]}}, "srt")
    branches_hit.add("srt_negative")
    try:
        export({}, "bad")
    except ValueError:
        branches_hit.add("bad_fmt")
    print("手工分支计数:", sorted(branches_hit))
    print(f"覆盖 {len(branches_hit)}/4 分支")
    assert len(branches_hit) == 4
    print("回退度量通过：即使无 coverage 包，也能显式验证分支覆盖")
# 预期输出（有 coverage 时）:
# has_coverage: True
# coverage report ... 包含行覆盖百分比与缺失行
# branch coverage enabled: True
# （无 coverage 时）:
# has_coverage: False
# 手工分支计数: ['bad_fmt', 'srt_negative', 'txt_empty_fallback', 'txt_normal']
# 覆盖 4/4 分支
```

文件 `book/part1_软件工程筑基/chapter02_构筑代码质量的护城河/demo_quality_gate.py`（把门禁写进脚本，本地即可校验）：

```{code-cell} ipython3
# 文件 book/part1_软件工程筑基/chapter02_构筑代码质量的护城河/demo_quality_gate.py
import subprocess, sys, pathlib, tempfile, os

# 模拟 CI 门禁脚本：依次执行 ruff / mypy / pytest，任一失败即非零退出
# 此处为演示，仅对 m2t 的轻量检查；真实 CI 会加入 --cov-fail-under
checks = []

def run(cmd: list[str], label: str):
    result = subprocess.run(cmd, capture_output=True, text=True)
    ok = result.returncode == 0
    print(f"[{label}] exit={result.returncode} {'PASS' if ok else 'FAIL'}")
    if not ok:
        print((result.stdout + result.stderr)[:800])
    checks.append(ok)
    return ok

# 1) ruff
run([sys.executable, "-m", "ruff", "check", "m2t/"], "ruff check")
# 2) mypy（非严格，教学仓默认值）
run([sys.executable, "-m", "mypy", "m2t/"], "mypy")
# 3) pytest 快速体感（若无用例则仅验证可执行）
try:
    import pytest  # type: ignore
    has_pytest = True
except ImportError:
    has_pytest = False
print("has_pytest:", has_pytest)
if has_pytest:
    # 构造一个临时 pytest 用例，验证 store 的 hermetic 测试可通过
    with tempfile.TemporaryDirectory() as td:
        test_file = pathlib.Path(td) / "test_gate.py"
        test_file.write_text(
            "from m2t.store import TaskStore\n"
            "import tempfile, pathlib\n"
            "def test_create():\n"
            "    import tempfile, pathlib\n"
            "    from m2t.store import TaskStore\n"
            "    import tempfile\n"
            "    with tempfile.TemporaryDirectory() as td:\n"
            "        s = TaskStore(pathlib.Path(td) / 't.db')\n"
            "        s.create('x', 'a.wav')\n"
            "        assert s.get('x')['filename'] == 'a.wav'\n",
            encoding="utf-8",
        )
        r = subprocess.run([sys.executable, "-m", "pytest", "-q", str(test_file)], capture_output=True, text=True)
        print("pytest gate:", "PASS" if r.returncode == 0 else "FAIL")
        print((r.stdout + r.stderr)[:800])
        checks.append(r.returncode == 0)

print("门禁汇总:", "全部通过" if all(checks) else "存在失败（CI 将阻断合并）")
print("启示：门禁是‘本地与 CI 一致’的自动化脚本，而非口头约定")
# 预期输出:
# [ruff check] exit=0 PASS
# [mypy] exit=0 PASS (或 1 若有类型告警，需在提交前修复)
# has_pytest: True
# pytest gate: PASS
# 门禁汇总: 全部通过
```
